# Update de la carte
sudo apt update
sudo apt full-upgrade -y
sudo reboot

# Installation firmware st7796s
sudo install -m 644 st7796s.bin /lib/firmware/st7796s.bin

# Réglage SPI (Interface > SPI > Enable)
sudo raspi-config

# Ajouter des configs dans /boot/firmware/config.txt
sudo nano /boot/firmware/config.txt
"""
dtparam=spi=on
dtoverlay=mipi-dbi-spi,speed=16000000
dtparam=compatible=st7796s\0panel-mipi-dbi-spi
dtparam=reset-gpio=27,dc-gpio=25,backlight-gpio=18
"""

# Activation de l'écran (à chaque boot)
sudo sh -c 'echo 0 > /sys/class/graphics/fb1/blank'

# Affichage d'une image (auto dimensionné)
sudo fbi -T 1 -d /dev/fb1 -a -noverbose image.png

# Effacer le framebuffer (clean l'écran)
sudo dd if=/dev/zero of=/dev/fb1

# Activation rétro éclairage
sudo pinctrl set 18 op dh

# Vérification version système
uname -a
